cmake_minimum_required(VERSION 3.24)
project(MfgRemixChecks LANGUAGES CXX)
set(MFG_SOURCE "" CACHE PATH "Pinned MFG checkout root")
set(MFG_INPUTS "" CACHE PATH "Locally recovered and verified inputs")
set(MFG_DLL "" CACHE FILEPATH "Locally built candidate")
set(REMIX_DLL "" CACHE FILEPATH "Pinned installed Remix runtime")
set(OLD_REMIX_DLL "" CACHE FILEPATH "Previously validated 71a581da runtime for profile regression checks")
set(NATIVE "${MFG_SOURCE}/source/native")
if(MFG_INPUTS AND EXISTS "${MFG_INPUTS}/validate_recovered.cpp")
    add_executable(MfgRecoveredContracts "${MFG_INPUTS}/validate_recovered.cpp" "${NATIVE}/ampere_native_cache.cpp")
endif()
add_executable(MfgRemixHostSmoke remix_host_smoke.cpp)
add_library(MfgNoRemixApi MODULE remix_no_api_test.cpp)
set_target_properties(MfgNoRemixApi PROPERTIES PREFIX "" OUTPUT_NAME "d3d9")
# The API-shaped impostor must live in a separate output folder, because
# both fixtures intentionally use the Remix runtime's d3d9.dll basename.
add_library(MfgUnknownRemixApi MODULE remix_no_api_test.cpp)
set_target_properties(MfgUnknownRemixApi PROPERTIES PREFIX "" OUTPUT_NAME "d3d9"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/unknown-remix-api")
target_compile_definitions(MfgUnknownRemixApi PRIVATE MFG_TEST_EXPORT_FAKE_REMIX_API=1)
add_executable(MfgRuntimeProfileTests remix_runtime_profile_test.cpp)
add_executable(MfgRemixPresentSmoke remix_present_smoke.cpp)
add_executable(OptiRemixProbe optiscaler_probe.cpp)
add_executable(RemixNeuralProbe runtime_neural_probe.cpp)
add_library(MfgJavaHostProbe SHARED optiscaler_probe.cpp)
target_compile_definitions(MfgJavaHostProbe PRIVATE MFG_PROBE_JAVA_HOST=1)
target_include_directories(MfgRemixPresentSmoke PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../rtxremix-java/native/include")
target_link_libraries(MfgRemixPresentSmoke PRIVATE user32)
target_include_directories(OptiRemixProbe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../rtxremix-java/native/include")
target_link_libraries(OptiRemixProbe PRIVATE user32)
target_include_directories(RemixNeuralProbe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../rtxremix-java/native/include")
target_link_libraries(RemixNeuralProbe PRIVATE user32)
target_compile_features(RemixNeuralProbe PRIVATE cxx_std_20)
target_compile_definitions(RemixNeuralProbe PRIVATE UNICODE _UNICODE NOMINMAX)
target_include_directories(MfgJavaHostProbe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../rtxremix-java/native/include")
target_link_libraries(MfgJavaHostProbe PRIVATE user32)
set(MFG_CHECK_TARGETS MfgRemixHostSmoke MfgNoRemixApi MfgUnknownRemixApi MfgRuntimeProfileTests MfgRemixPresentSmoke OptiRemixProbe MfgJavaHostProbe)
if(TARGET MfgRecoveredContracts)
    list(APPEND MFG_CHECK_TARGETS MfgRecoveredContracts)
endif()
foreach(target IN LISTS MFG_CHECK_TARGETS)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE "${NATIVE}")
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX)
endforeach()
if(TARGET MfgRecoveredContracts)
    target_link_libraries(MfgRecoveredContracts PRIVATE bcrypt)
endif()
target_link_libraries(MfgRuntimeProfileTests PRIVATE bcrypt)
enable_testing()
if(TARGET MfgRecoveredContracts)
    add_test(NAME recovered_contracts COMMAND MfgRecoveredContracts "${MFG_INPUTS}")
endif()
if(MFG_DLL AND REMIX_DLL)
    add_test(NAME remix_host_registration COMMAND MfgRemixHostSmoke "${MFG_DLL}" "${REMIX_DLL}" "$<TARGET_FILE:MfgNoRemixApi>" "$<TARGET_FILE:MfgUnknownRemixApi>")
endif()
if(REMIX_DLL)
    # The v0.1.1 identity/architecture regression must not silently disappear
    # when only the current runtime is available. Legacy coverage is additive.
    if(OLD_REMIX_DLL)
        add_test(NAME remix_runtime_profiles COMMAND MfgRuntimeProfileTests "${OLD_REMIX_DLL}" "${REMIX_DLL}")
    else()
        add_test(NAME remix_runtime_profiles COMMAND MfgRuntimeProfileTests "${REMIX_DLL}")
    endif()
endif()
